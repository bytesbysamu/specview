# 🛠️ Task 1: User model — builder + principles

**Purpose**: Extend `superapp_users` with `builder` and `principles` JSONB columns so every subsequent AI feature (spec chain in Task 2, onboarding in Task 5, photoshoot retrofit in Task 6) can read accumulated user context. Foundation task — everything downstream reads from here.

**Effort**: 1 day

**Dependencies**: None

**Parallel With**: —

**Blocks**: Task 2 (chain primitive reads `user.builder` / `user.principles_for(feature)`), Task 5 (onboarding writes `user.builder`).

**Related**:
- [Solution Architecture](./architecture.md)
- [Epic](./epic.md)

## 1. Context

This task lays the data foundation every downstream AI feature depends on. Today `superapp_users` stores only auth primitives — email plus a magic-link UUID — which means a chain primitive (Task 2) would have no place to read "who is this builder" or "what decisions has this user accumulated for *this* feature". Without that, every prompt step re-asks questions it should already know. We add two JSONB columns: `builder` (one mutable profile blob, nullable — NULL is the signal onboarding hasn't run) and `principles` (namespaced per feature, shape `{photoshoot: {...}, spec: {...}}`, NOT NULL with default `'{}'` so read sites never null-check). We update the `User` SQLModel with typed accessors (`builder_profile()`, `principles_for(feature)`), regenerate the OpenAPI-driven user schema so `GET /api/user/me` exposes both fields, and leave existing users untouched — no backfill, existing rows get NULL builder and empty principles.

**Trade-offs considered**:
- **Two columns vs. one merged JSONB**: kept separate so the mutable-but-stable profile and the append-only decision log can evolve independently without field collisions.
- **`principles` NOT NULL + server default `'{}'` vs. nullable**: default simplifies every read site (`u.principles_for(x)` never needs a None check). `builder` stays nullable because NULL carries load-bearing meaning (not yet onboarded).
- **Typed `BuilderProfile` dataclass vs. raw `Dict[str, Any]`**: typed accessor on the model for call-site autocomplete; the stored dict shape stays loose so we don't lock the schema before Task 5 finalises onboarding fields.

**Action**: Create the migration adding both columns. Use `sa.JSON` — SQLAlchemy 2.x maps this to JSONB on Postgres and JSON on SQLite, consistent with the "SQLite test portability" pattern in codebase.md.

**File**: `server/migrations/versions/20260417_add_builder_principles.py` (new)

**Pattern**:
```python
"""add builder and principles to superapp_users

Revision ID: 20260417_add_builder_principles
Revises: 20260416_add_original_image_url
Create Date: 2026-04-17
"""
from alembic import op
import sqlalchemy as sa

revision = "20260417_add_builder_principles"
down_revision = "20260416_add_original_image_url"  # confirmed in Pre-flight
branch_labels = None
depends_on = None


def upgrade() -> None:
    op.add_column(
        "superapp_users",
        sa.Column("builder", sa.JSON(), nullable=True),
    )
    op.add_column(
        "superapp_users",
        sa.Column(
            "principles",
            sa.JSON(),
            nullable=False,
            server_default=sa.text("'{}'"),
        ),
    )


def downgrade() -> None:
    op.drop_column("superapp_users", "principles")
    op.drop_column("superapp_users", "builder")
```

### Step 3: Extend the `User` SQLModel with columns + typed accessors

**Action**: Add `builder`, `principles` columns and `builder_profile()`, `principles_for(feature)` accessors. Keep `principles` loose-typed (`Dict[str, Dict[str, Any]]`).

**File**: `server/modules/user/model.py` (or whatever Step 1 recorded)

**Pattern**:
```python
from __future__ import annotations
from dataclasses import dataclass, fields
from typing import Any, Dict, Optional
from sqlalchemy import JSON
from sqlalchemy.orm import Mapped, mapped_column
# ... existing imports + Base ...


@dataclass
class BuilderProfile:
    name: Optional[str] = None
    role: Optional[str] = None
    stack: Optional[list[str]] = None
    style: Optional[str] = None
    goals: Optional[str] = None

    @classmethod
    def from_dict(cls, data: Optional[Dict[str, Any]]) -> Optional["BuilderProfile"]:
        if data is None:
            return None
        known = {f.name for f in fields(cls)}
        return cls(**{k: v for k, v in data.items() if k in known})


class User(Base):
    __tablename__ = "superapp_users"
    # ... existing columns unchanged ...
    builder: Mapped[Optional[Dict[str, Any]]] = mapped_column(JSON, nullable=True)
    principles: Mapped[Dict[str, Dict[str, Any]]] = mapped_column(
        JSON, nullable=False, server_default="{}", default=dict,
    )

    def builder_profile(self) -> Optional[BuilderProfile]:
        return BuilderProfile.from_dict(self.builder)

    def principles_for(self, feature: str) -> Dict[str, Any]:
        return (self.principles or {}).get(feature, {})
```

# server/tests/test_user_model.py
import uuid
from modules.user.model import User, BuilderProfile


def test_newUser_principlesDefaultsToEmptyDict(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co")
    db_session.add(u); db_session.commit(); db_session.refresh(u)
    assert u.principles == {}, "NOT NULL + server_default '{}' must yield empty dict"


def test_newUser_builderIsNullByDefault(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co")
    db_session.add(u); db_session.commit(); db_session.refresh(u)
    assert u.builder is None, "NULL builder is the onboarding-needed signal"


def test_builderProfile_returnsTypedInstance_whenBuilderPopulated(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co",
             builder={"name": "Sam", "stack": ["angular", "flask"], "goals": "ship"})
    db_session.add(u); db_session.commit()
    bp = u.builder_profile()
    assert isinstance(bp, BuilderProfile)
    assert bp.name == "Sam"
    assert bp.stack == ["angular", "flask"]
    assert bp.goals == "ship"
    assert bp.role is None and bp.style is None


def test_builderProfile_returnsNone_whenBuilderNull(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co")
    db_session.add(u); db_session.commit()
    assert u.builder_profile() is None


def test_builderProfile_ignoresUnknownKeys(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co",
             builder={"name": "Sam", "__experimental__": True, "xyz": 1})
    db_session.add(u); db_session.commit()
    bp = u.builder_profile()
    assert bp.name == "Sam", "known keys preserved"
    assert not hasattr(bp, "__experimental__"), "unknown keys must not leak onto dataclass"


def test_principlesFor_returnsEmptyDict_whenNamespaceAbsent(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co", principles={"photoshoot": {"ratio": "3:4"}})
    db_session.add(u); db_session.commit()
    assert u.principles_for("spec") == {}
    assert u.principles_for("photoshoot") == {"ratio": "3:4"}


def test_principlesFor_namespacesAreIndependent(db_session):
    u = User(id=uuid.uuid4(), email="a@b.co",
             principles={"photoshoot": {"a": 1}, "spec": {"b": 2}})
    db_session.add(u); db_session.commit()
    assert u.principles_for("photoshoot") == {"a": 1}
    assert u.principles_for("spec") == {"b": 2}

# server/tests/test_user_migration.py
from sqlalchemy import create_engine, inspect
from alembic import command
from alembic.config import Config


def _cfg(url: str) -> Config:
    cfg = Config("server/alembic.ini")
    cfg.set_main_option("sqlalchemy.url", url)
    return cfg


def test_migration_upgradeAddsBothColumns(tmp_path):
    url = f"sqlite:///{tmp_path}/mig.db"
    command.upgrade(_cfg(url), "head")
    cols = {c["name"] for c in inspect(create_engine(url)).get_columns("superapp_users")}
    assert "builder" in cols
    assert "principles" in cols


def test_migration_downgradeRemovesBothColumns(tmp_path):
    url = f"sqlite:///{tmp_path}/mig.db"
    cfg = _cfg(url)
    command.upgrade(cfg, "head")
    command.downgrade(cfg, "-1")
    cols = {c["name"] for c in inspect(create_engine(url)).get_columns("superapp_users")}
    assert "builder" not in cols
    assert "principles" not in cols
